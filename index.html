import os from reportlab.lib.pagesizes import A4, landscape from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle, Image from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle from reportlab.lib import colors # Exemplo de função que gera o PDF no seu app.py: def gerar_pdf_bradesco(dados, socios, caminho_saida="bradesco_net_empresa.pdf"): # Cria o documento em paisagem (A4 landscape) doc = SimpleDocTemplate( caminho_saida, pagesize=landscape(A4), rightMargin=30, leftMargin=30, topMargin=30, bottomMargin=30 ) story = [] # --- CABEÇALHO COM A LOGO REAL --- logo_path = "brada.png" # Certifique-se de que a imagem brada.png está na mesma pasta elementos_cabecalho = [] if os.path.exists(logo_path): # Insere a imagem da logo com tamanho proporcional (ex: 38x38 pixels) img_logo = Image(logo_path, width=38, height=38) elementos_cabecalho.append(img_logo) # Título ao lado da logo style_titulo = ParagraphStyle( 'TituloHeader', fontName='Helvetica-Bold', fontSize=16, textColor=colors.HexColor('#FFFFFF') ) elementos_cabecalho.append(Paragraph("Bradesco Net Empresa", style_titulo)) # Tabela do Cabeçalho (Logo + Título) tabela_header = Table([elementos_cabecalho], colWidths=[50, None]) tabela_header.setStyle(TableStyle([ ('VALIGN', (0,0), (-1,-1), 'MIDDLE'), ('BOTTOMPADDING', (0,0), (-1,-1), 10), ])) story.append(tabela_header) # Adicione aqui o restante dos dados do seu relatório (matrícula, ag, conta, tabela de sócios, etc.) doc.build(story)
